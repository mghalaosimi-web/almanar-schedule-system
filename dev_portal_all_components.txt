function TenantContextBar({ tenants, selected, onChange, isAr }) {
  const { universities = [], colleges = [] } = tenants;
  const allTenants = [
    { id: 'ALL', label: isAr ? 'كافة الجامعات والكليات' : 'All Tenants (Global)', type: 'ALL', icon: '🌐' },
    ...universities.map(u => ({ id: `UNI_${u.id}`, label: u.name, type: 'UNIVERSITY', universityId: u.id, icon: '🏛️' })),
    ...colleges.map(c => ({
      id: `COL_${c.id}`,
      label: `${c.name}${c.university ? ` — ${c.university.name}` : ''}`,
      type: 'COLLEGE', collegeId: c.id, universityId: c.universityId, icon: '🏫'
    })),
  ];

  const current = allTenants.find(t => t.id === selected?.id) || allTenants[0];

  return (
    <div className="flex items-center gap-2">
      <span className="text-[9px] font-bold text-white/50 uppercase tracking-wider hidden md:block">
        {isAr ? 'السياق:' : 'Context:'}
      </span>
      <div className="relative">
        <select
          value={current.id}
          onChange={e => {
            const found = allTenants.find(t => t.id === e.target.value);
            onChange(found || allTenants[0]);
          }}
          className="appearance-none bg-black/60 border border-white/10 hover:border-[var(--accent)]/40 text-white rounded-xl pl-7 pr-8 py-2 text-xs font-bold focus:outline-none cursor-pointer min-w-[200px]"
        >
          {allTenants.map(t => (
            <option key={t.id} value={t.id}>
              {t.icon} {t.label}
            </option>
          ))}
        </select>
      </div>
    </div>
  );
}

export default function DevPortal() {
  const { i18n } = useTranslation();
  const isAr = i18n.language === 'ar';
  const token = localStorage.getItem('manar_token');

  const [isUnlocked, setIsUnlocked] = useState(() => sessionStorage.getItem('manar_dev_unlocked') === 'true');
  const [passcode, setPasscode] = useState('');
  const [verifying, setVerifying] = useState(false);
  const [passcodeError, setPasscodeError] = useState('');

  const handleVerifyPasscode = async (e) => {
    e.preventDefault();
    setVerifying(true);
    setPasscodeError('');
    try {
      const res = await axios.post(`${API_URL}/api/admin/dev/verify-key`, 
        { passcode },
        { headers: { Authorization: `Bearer ${token}` } }
      );
      if (res.data?.success) {
        sessionStorage.setItem('manar_dev_unlocked', 'true');
        setIsUnlocked(true);
        toast.success(isAr ? 'تم فتح الوصول بنجاح' : 'Access granted successfully');
      }
    } catch (err) {
      setPasscodeError(err.response?.data?.error || (isAr ? 'رمز المرور غير صحيح' : 'Incorrect passcode'));
    } finally {
      setVerifying(false);
    }
  };

  const [sidebarExpanded, setSidebarExpanded] = useState(true);
  const [activeTab, setActiveTab]             = useState('telemetry');
  const [tenantContext, setTenantContext]     = useState({ id: 'ALL', type: 'ALL', label: 'All Tenants' });
  const [telemetry, setTelemetry]             = useState(null);

  if (!isUnlocked) {
    return (
      <div dir={isAr ? 'rtl' : 'ltr'} className="min-h-screen w-full bg-[var(--bg-primary)] text-slate-100 flex items-center justify-center p-6">
        <form onSubmit={handleVerifyPasscode} className="space-y-4 max-w-md w-full bg-black/45 p-8 rounded-3xl border border-white/10 text-center">
          <h2 className="text-xl font-black text-white">{isAr ? 'بوابة العبور الآمنة للمطور' : 'Developer Gateway'}</h2>
          <input 
            type="password"
            required
            value={passcode}
            onChange={e => setPasscode(e.target.value)}
            placeholder="••••••••••••"
            className="w-full text-center px-5 py-4 rounded-xl bg-white/5 border border-white/10 text-white font-mono text-lg"
          />
          <button type="submit" className="w-full py-4 rounded-xl bg-gradient-to-r from-cyan-500 to-purple-600 font-black text-xs uppercase">
            {isAr ? 'تحقق وعبور ⚡' : 'Authorize & Enter ⚡'}
          </button>
        </form>
      </div>
    );
  }

  return (
    <div dir={isAr ? 'rtl' : 'ltr'} className="min-h-screen bg-[var(--bg-primary)] text-slate-100 flex font-sans">
      <div className="p-6 w-full">
        <h1 className="text-base font-black text-white">{isAr ? 'مركز التحكم والتشخيص للمطورين' : 'Developer Command Center'}</h1>
      </div>
    </div>
  );
}
